CREATE TABLE 고객 (
  고객아이디 VARCHAR(20) PRIMARY KEY,
  이름 VARCHAR(10) NOT NULL,
  나이 INT,
  등급 VARCHAR(10) DEFAULT 'normal',
  직업 VARCHAR(10) DEFAULT '',
  적립금 INT DEFAULT 0
);


CREATE TABLE 제품 (
  제품번호 CHAR(3) PRIMARY KEY,
  제품명 VARCHAR(20),
  재고량 INT,
  단가 INT
);
ALTER TABLE 제품 ADD 제조업체 VARCHAR(20);


CREATE TABLE 주문 (
  주문번호 CHAR(3),
  주문고객 VARCHAR(20),
  주문제품 CHAR(3),
  수량 INT,
  배송지 VARCHAR(30),
  주문일자 DATETIME,
  FOREIGN KEY (주문고객) REFERENCES 고객(고객아이디)
      ON DELETE CASCADE,
  FOREIGN KEY (주문제품) REFERENCES 제품(제품번호)
      ON UPDATE CASCADE,
  PRIMARY KEY (주문번호)
);
