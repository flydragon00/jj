    "def check_disk_usage(path=\"/\", threshold=80):\n",

    "  usage = used / total * 100\n",
    "  return usage, threshold\n",

    "  msg[\"To\"] = RECIPIENT\n",

    "\n",
    "  msg[\"Subject\"] = subject\n",
